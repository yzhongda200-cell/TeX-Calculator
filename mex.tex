\long\def\mex#1{%
	\tcalc@ifmt{#1}
		{0}
		{\expandafter\tcalc@mex@sort@reverse\expandafter{\romannumeral-`?#1}}%
}
\long\def\tcalc@mex#1\tcalc@relax#2{%
	\tcalc@ifx\tcalc@mex@endignore{#2}
		{\tcalc@mex@reverse\tcalc@relax#1\tcalc@relax}
		{\tcalc@mex@ifneq{#1}{#2}
			{\tcalc@mex@ignore{\tcalc@mex@reverse\tcalc@relax#1\tcalc@relax}}
			{\expandafter\tcalc@mex\romannumeral-`?\tcalc@mex@step\tcalc@relax#1\tcalc@relax}}%
}
\def\tcalc@mex@step#1\tcalc@relax#2{%
	\tcalc@ifx#2\tcalc@relax
		{#11\tcalc@relax}
		{\tcalc@ifx9#2
			{\tcalc@mex@step#10\tcalc@relax}
			{\csname tcalc@mex@add@\string#2\endcsname{#1}}}%
}
\def\tcalc@mex@reverse#1\tcalc@relax#2{%
	\tcalc@ifx\tcalc@relax{#2}
		{#1}
		{\tcalc@ifx{ }{#2}
			{\tcalc@mex@reverse#1\tcalc@relax}
			{\tcalc@mex@reverse#2#1\tcalc@relax}}%
}
\def\tcalc@mex@ifneq#1#2{%
	\tcalc@mex@ifneq@a#1\tcalc@relax#2\tcalc@relax
}
\def\tcalc@mex@ifneq@a#1#2\tcalc@relax#3#4\tcalc@relax{%
	\tcalc@ifmt{#2}
		{\tcalc@ifmt{#4}
			{\tcalc@ifx#1#3
				{\@secondoftwo}
				{\@firstoftwo}}
			{\@firstoftwo}}
		{\tcalc@ifmt{#4}
			{\@firstoftwo}
			{\tcalc@ifx#1#3
				{\tcalc@mex@ifneq@a#2\tcalc@relax#4\tcalc@relax}
				{\@firstoftwo}}}%
}
\def\tcalc@mex@sort@reverse#1{%
	\expandafter\tcalc@ifmt\expandafter{\romannumeral-`?#1}
		{0}
		{\tcalc@mex@sort@reverse@a\tcalc@relax\tcalc@relax
			#1\tcalc@relax}%
}
\def\tcalc@mex@sort@reverse@a#1\tcalc@relax#2\tcalc@relax#3{%
	\tcalc@ifx\tcalc@relax{#3}
		{\tcalc@ifmt{#1}
			{\tcalc@ifmt{#2}
				{0}
				{\tcalc@mex@sort#2\tcalc@relax\tcalc@mex@endignore}}
			{\tcalc@mex@sort@reverse@a\tcalc@relax#1|#2\tcalc@relax{#3}}}
		{\expandafter\tcalc@ifmt\expandafter{\romannumeral-`?#3}
			{\tcalc@mex@sort@reverse@a#1\tcalc@relax#2\tcalc@relax}
			{\tcalc@ifmt{#1}
				{\expandafter\tcalc@mex@sort@reverse@a\romannumeral-`?\expandafter\tcalc@mex@reverse\expandafter\tcalc@relax\romannumeral-`?#3\tcalc@relax\tcalc@relax#2\tcalc@relax}
				{\tcalc@mex@sort@reverse@a\tcalc@relax#1|#2\tcalc@relax{#3}}}}%
}

\def\tcalc@mex@sort@iflt#1#2{%

\def\tcalc@mex@sort@iflt@a#1#2\tcalc@relax#3#4\tcalc@relax{%
	\tcalc@ifmt{#2#4}
		{\ifnum#1<#3 \expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi}
		{\ifnum#1<#3 \expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
			{\@firstoftwo}
			{\ifnum#3<#1 \expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
				{\@secondoftwo}
				{\tcalc@mex@sort@iflt@a#2\tcalc@relax#4\tcalc@relax}}}%
}
